const fs=require('fs'),path=require('path'),assert=require('assert'),crypto=require('crypto');
module.exports=async function({contexts,db,origin,root}){
 const evidence={startedAt:new Date().toISOString(),checks:[],scope:'One isolated article lifecycle through real authenticated APIs; not full CMS/manual/publication UAT'};
 const record=(name,status,details)=>{evidence.checks.push({name,status,...details});console.log('Editorial QA: '+name+' '+status)};
 async function call(role,method,url,data,expected,label){const response=await contexts[role].request.fetch(origin+url,{method,data,headers:{Origin:origin},timeout:30000});const payload=await response.json();assert.equal(response.status(),expected,label+': '+JSON.stringify(payload));record(label,'passed',{httpStatus:response.status()});return payload}
 const original=await db.collection('articles').findOne();assert(original);const beforeViews=original.views,beforeUpdated=original.updatedAt;
 for(let i=0;i<3;i++){const response=await contexts.reporter.request.get(origin+'/api/v1/public/articles/'+original.slug);assert.equal(response.status(),200)}const after=await db.collection('articles').findOne({_id:original._id});assert.equal(after.views,beforeViews);assert.deepEqual(after.updatedAt,beforeUpdated);record('Three public GETs preserve article views and updatedAt','passed',{});
 const slug='qa-r06-editorial-'+crypto.randomBytes(4).toString('hex');const payload={title:'R06 isolated editorial acceptance article',summary:'This isolated QA article exercises newsroom role handoff, approval and release without writing to any production database.',content:'<h2>Isolated newsroom acceptance</h2><p>'+('This is a disposable QA article for verifying the real authenticated editorial workflow, ownership, assignment and publication visibility. ').repeat(10)+'</p>',image:original.image,category:original.category,author:'LokSwami QA Reporter',slug,sourceType:'direct',isBreaking:false,isTrending:false,editorial:{storyType:'standard',factCheckStatus:'verified',headlineSupportConfirmed:true,duplicateCheckComplete:true},reporterMeta:{locationTag:'Indore QA isolation',sourceInfo:'Disposable isolated acceptance fixture; never production news.'},seo:{metaTitle:'R06 isolated editorial acceptance article',metaDescription:'Disposable QA fixture to validate real newsroom permissions and visibility.',focusKeyword:'QA'},intent:'draft'};
 await call('reporter','POST','/api/admin/articles',{...payload,intent:'publish'},403,'Reporter cannot publish directly');
 const created=await call('reporter','POST','/api/admin/articles',payload,201,'Reporter creates owned draft');let article=created.data;const id=String(article._id||article.id);assert(id);assert.equal(article.workflow.status,'draft');evidence.qaArticleId=id;
 async function visible(expected,label){const response=await contexts.reporter.request.get(origin+'/api/v1/public/articles/'+slug);assert.equal(response.status(),expected);record(label,'passed',{httpStatus:response.status()})}
 await visible(404,'Draft excluded from public reader');
 async function action(role,name,expectedStatus,extra={}){const updated=await call(role,'PATCH','/api/admin/articles/'+id,{action:name,expectedVersion:article.version,...extra},200,role+' '+name);article=updated.data;assert.equal(article.workflow.status,expectedStatus);}
 await action('reporter','submit','submitted');
 const copy=await db.collection('users').findOne({loginId:'qa-copy-editor'});await action('admin','assign','assigned',{assignedToId:String(copy._id)});
 await call('reporter','PATCH','/api/admin/articles/'+id,{action:'approve',expectedVersion:article.version},403,'Reporter approval denied');
 await action('copy_editor','start_review','in_review');await action('copy_editor','move_to_copy_edit','copy_edit');await action('copy_editor','mark_ready_for_approval','ready_for_approval');
 await call('copy_editor','PATCH','/api/admin/articles/'+id,{action:'approve',expectedVersion:article.version},403,'Copy Editor approval denied');
 await action('admin','approve','approved');await action('admin','schedule','scheduled',{scheduledFor:new Date(Date.now()+3600000).toISOString()});await visible(404,'Future scheduled article excluded from reader');
 await action('admin','publish','published');await visible(200,'Published article appears in reader');await action('super_admin','archive','archived');await visible(404,'Archived article excluded from reader');
 evidence.completedAt=new Date().toISOString();evidence.ok=true;fs.writeFileSync(path.join(root,'qa-runtime/r06-editorial-report.json'),JSON.stringify(evidence,null,2));return evidence;
};
