const fs=require('fs'),path=require('path'),assert=require('assert'),crypto=require('crypto');
module.exports=async function({contexts,db,origin,root,helperSourceSha256}){
 const config=JSON.parse(fs.readFileSync(path.join(root,'qa-runtime/hqs-private-config.json'))),req=require('module').createRequire(path.join(config.repo,'package.json'));
 const {chromium}=req('@playwright/test');
 const report={startedAt:new Date().toISOString(),helperSourceSha256,scope:'Work Queue timestamp hydration only: four real roles, Hindi/English, India/US browser timezones. No complete CMS acceptance claim.',seedChecks:[],cases:[],pageErrors:[]};
 const save=()=>fs.writeFileSync(path.join(root,'qa-runtime/hqs-editorial-report.json'),JSON.stringify(report,null,2));
 async function call(role,url,method,data){const r=await contexts[role].request.fetch(origin+url,{method,data,headers:{Origin:origin}});const p=await r.json();assert(r.ok(),role+' '+url+' '+r.status()+' '+JSON.stringify(p));report.seedChecks.push({role,url,method,httpStatus:r.status()});save();return p.data}
 const suffix=crypto.randomBytes(4).toString('hex'),articles={};
 for(const u of config.users){articles[u.role]=await call(u.role,'/api/admin/articles','POST',{title:'Timezone QA '+u.role,summary:'Explicit synthetic local QA article for verifying Work Queue timestamps across browser timezones and languages.',content:'<p>'+('This is an explicitly synthetic isolated QA article for real newsroom acceptance, timestamp rendering, role restrictions and browser hydration. ').repeat(10)+'</p>',image:'/logo-app-192.png',category:'general',author:'Synthetic QA',slug:'qa-work-clock-'+u.role.replaceAll('_','-')+'-'+suffix,sourceType:'direct',intent:'draft',isBreaking:false,isTrending:false,editorial:{storyType:'standard',factCheckStatus:'verified',headlineSupportConfirmed:true,duplicateCheckComplete:true},reporterMeta:{locationTag:'Indore QA',sourceInfo:'Synthetic isolated acceptance fixture'},seo:{metaTitle:'Timezone QA article',metaDescription:'Synthetic isolated timestamp acceptance fixture',focusKeyword:'QA'}})}
 const reporterId=String(articles.reporter._id||articles.reporter.id);let article=await call('reporter','/api/admin/articles/'+reporterId,'PATCH',{action:'submit',expectedVersion:articles.reporter.version});const copy=await db.collection('users').findOne({loginId:'qa-copy-editor'});await call('admin','/api/admin/articles/'+reporterId,'PATCH',{action:'assign',expectedVersion:article.version,assignedToId:String(copy._id),dueAt:new Date(Date.now()+86400000).toISOString()});
 const adminId=String(articles.admin._id||articles.admin.id);await call('admin','/api/admin/articles/'+adminId,'PATCH',{action:'schedule',expectedVersion:articles.admin.version,scheduledFor:new Date(Date.now()+3600000).toISOString()});
 const routes=[['/admin/work',['super_admin','admin','copy_editor','reporter']],['/admin/my-work',['admin','copy_editor','reporter']],['/admin/review-queue',['super_admin','admin']],['/admin/assignments',['super_admin','admin']],['/admin/content-queue',['super_admin','admin']]];
 const browser=await chromium.launch({headless:true,executablePath:path.join(root,'qa-runtime/browser/chrome-headless-shell-linux64/chrome-headless-shell')});
 try{
  for(const timezoneId of ['Asia/Kolkata','America/Los_Angeles'])for(const language of ['hi','en'])for(const user of config.users){
   const context=await browser.newContext({viewport:{width:1440,height:1000},timezoneId});await context.route('**/*',r=>new URL(r.request().url()).origin===origin?r.continue():r.abort('blockedbyclient'));
   await context.addInitScript(()=>{window.__hqsDocumentErrors=[];window.addEventListener('error',e=>window.__hqsDocumentErrors.push({url:location.href,message:e.message,stack:e.error?.stack}))});const page=await context.newPage();
   page.on('pageerror',e=>{report.pageErrors.push({role:user.role,language,timezoneId,path:new URL(page.url()).pathname,message:e.message});save()});
   await page.goto(origin+'/signin?redirect=/admin/work',{waitUntil:'domcontentloaded'});await page.waitForFunction(()=>Array.from(document.querySelectorAll('button')).some(e=>e.textContent.includes('Reader & Subscriber')&&Object.keys(e).some(k=>k.startsWith('__reactProps'))));await page.getByRole('button',{name:'Reader & Subscriber',exact:true}).click();await page.getByRole('button',{name:'Newsroom Team',exact:true}).click();await page.getByRole('textbox',{name:'Staff Login ID or Email',exact:true}).fill(user.loginId);await page.locator('input[autocomplete="current-password"]:visible').fill(user.password);await page.getByRole('button',{name:'Sign in to Newsroom',exact:true}).click();await page.waitForURL(u=>u.pathname.startsWith('/admin'),{timeout:60000});
   const session=await (await context.request.get(origin+'/api/auth/session')).json();assert.equal(session.user.role,user.role);await page.waitForTimeout(300);
   const toggle=page.getByRole('button',{name:language==='en'?'Switch to English':'Switch to Hindi',exact:true});if(await toggle.count())await toggle.click();await page.waitForTimeout(100);
   assert.equal(await page.getByRole('button',{name:language==='en'?'Switch to Hindi':'Switch to English',exact:true}).count(),1,'Actual requested UI language must be selected');
   for(const [route,roles] of routes){const before=report.pageErrors.length;await page.goto(origin+route,{waitUntil:'domcontentloaded',timeout:45000});if(!roles.includes(user.role))await page.waitForURL(u=>u.pathname==='/admin/work'&&u.searchParams.get('access')==='denied',{timeout:15000});else assert.equal(new URL(page.url()).pathname,route);await page.waitForTimeout(500);const errors=await page.evaluate(()=>window.__hqsDocumentErrors);const dateRows=await page.locator('p').filter({hasText:' · Updated '}).allTextContents();assert(dateRows.length>0,'Actual seeded Work Queue timestamp rows must render');const dates=await db.collection('articles').find().toArray();const expectedDates=await page.evaluate(({dates,language})=>dates.map(d=>new Intl.DateTimeFormat(language==='hi'?'hi-IN':'en-IN',{timeZone:'Asia/Kolkata',day:'2-digit',month:'short',hour:'2-digit',minute:'2-digit'}).format(new Date(d))),{dates:dates.map(d=>d.updatedAt.toISOString()),language});assert(dateRows.every(row=>expectedDates.some(d=>row.includes(d))),'Displayed timestamp must match India time in the selected language');report.cases.push({role:user.role,language,timezoneId,requestedRoute:route,access:roles.includes(user.role)?'allowed':'denied',actualPath:new URL(page.url()).pathname,pageErrors:errors,eventErrorsAtDelivery:report.pageErrors.slice(before),timestampRows:dateRows.length,indiaTimestampAssertion:'passed'});save();console.log('HQS '+user.role+' '+language+' '+timezoneId+' '+route+' '+errors.length+' document errors')}
   await context.close();
  }
  report.overallRunOk=report.pageErrors.length===0;report.remainingObservedErrors=report.pageErrors;assert(report.cases.every(c=>c.pageErrors.length===0),'Target Work Queue documents must not contain hydration/runtime errors');report.targetScopeOk=true;report.ok=report.overallRunOk;report.completedAt=new Date().toISOString();save();return report;
 }finally{await browser.close()}
};
